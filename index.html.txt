                            <h3 className="text-sm font-extrabold text-[#292824] leading-snug">{item.thumbnailTitle}</h3>
                          </div>
                          <div className="flex items-center justify-between text-[11px] pt-2 border-t border-[#D5CFC4]">
                            <div className="flex gap-1 flex-wrap">
                              {item.activeChannels.map((chKey) => {
                                const b = getChannelBadgeInfo(chKey);
                                return <span key={chKey} className={`px-1.5 py-0.5 rounded text-[9px] font-bold border ${b.color}`}>{b.label}</span>;
                              })}
                            </div>
                            <span className="font-mono text-[10px] text-[#866143] font-bold">{item.duration}</span>
                          </div>
                        </div>
                        <div className="p-4">
                          <div className="flex justify-between text-[11px] text-[#866143] font-semibold mb-1">
                            <span>급상승 시장 키워드</span>
                            <span className="font-bold text-[#292824]">블루오션 {item.blueOceanScore}점</span>
                          </div>
                          <div className="text-base font-extrabold text-[#292824] mb-3">{item.keyword}</div>
                          <div className="p-3 rounded-xl bg-[#F2F1ED] border border-[#D5CFC4] space-y-1.5">
                            <div className="flex justify-between text-xs">
                              <span className="text-[#866143] font-bold">매칭 성과</span>
                              <span className="font-bold text-[#292824]">{item.platformSource}</span>
                            </div>
                            <div className="pt-1.5 border-t border-[#DEDAD2] text-xs">
                              <span className="text-[10px] font-bold text-[#866143] block">소비자 불만 1위 (빈틈 기회)</span>
                              <span className="text-[#292824] font-medium line-clamp-1">“{item.vocPainPoints[0].issue}”</span>
                            </div>
                          </div>
                        </div>
                      </div>
                      <div className="px-4 py-2.5 bg-[#E2DDD4]/60 border-t border-[#D5CFC4] flex justify-between items-center text-xs">
                        <span className="text-[#292824]/80 truncate">{item.format}</span>
                        <span className="font-extrabold text-[#866143] shrink-0">시장조사 열기 ›</span>
                      </div>
                    </article>
                  );
                })}
              </div>
            </main>
          </div>

          {/* 상세 리포트 모달 */}
          {activeRef && (
            <div onClick={() => setActiveRef(null)} className="fixed inset-0 z-50 bg-[#292824]/60 backdrop-blur-sm flex items-center justify-center p-3 sm:p-6 overflow-y-auto">
              <div onClick={(e) => e.stopPropagation()} className="bg-[#F2F1ED] border border-[#D5CFC4] rounded-3xl max-w-5xl w-full max-h-[90vh] overflow-y-auto p-5 sm:p-6 shadow-2xl">
                <div className="flex items-center justify-between pb-4 mb-5 border-b border-[#D5CFC4]">
                  <div>
                    <span className="text-xs font-extrabold text-[#866143]">{activeRef.categoryLabel} · {activeRef.growth} ▲</span>
                    <h2 className="text-lg sm:text-xl font-extrabold text-[#292824]">{activeRef.keyword}</h2>
                  </div>
                  <button onClick={() => setActiveRef(null)} className="px-3 py-1.5 rounded-xl bg-[#EAE6DF] border border-[#D5CFC4] text-xs font-bold text-[#292824]">닫기 ✕</button>
                </div>

                <div className="grid grid-cols-1 lg:grid-cols-12 gap-6">
                  {/* 좌측 후킹 구조 */}
                  <div className="lg:col-span-5 space-y-4">
                    <div className="p-4 rounded-2xl bg-[#EAE6DF] border border-[#D5CFC4]">
                      <div className="text-xs font-bold text-[#866143] mb-1">{activeRef.memeTag}</div>
                      <div className="text-base font-extrabold text-[#292824] mb-1">{activeRef.thumbnailTitle}</div>
                      <p className="text-xs text-[#292824]/80 mb-3">{activeRef.thumbnailSub}</p>
                      <div className="flex justify-between text-xs font-bold pt-2 border-t border-[#D5CFC4]">
                        <span>조회수 {activeRef.views}</span>
                        <span>저장 {activeRef.saves}</span>
                        <span>길이 {activeRef.duration}</span>
                      </div>
                    </div>
                    <div className="space-y-2">
                      <div className="text-xs font-extrabold text-[#866143]">숏폼 후킹 타임라인 해부</div>
                      {activeRef.hookStructure.map((h, i) => (
                        <div key={i} className="p-3 rounded-xl bg-[#EAE6DF] border border-[#D5CFC4]">
                          <div className="text-xs font-bold text-[#866143]">{h.step}</div>
                          <div className="text-xs text-[#292824] mt-0.5">{h.desc}</div>
                        </div>
                      ))}
                    </div>
                  </div>

                  {/* 우측 시장조사 탭 */}
                  <div className="lg:col-span-7 space-y-4">
                    <div className="grid grid-cols-4 gap-1.5 p-1 bg-[#EAE6DF] rounded-xl border border-[#D5CFC4]">
                      <button onClick={() => setActiveTab('market')} className={`py-2 rounded-lg text-xs font-bold ${activeTab === 'market' ? 'bg-[#866143] text-[#F2F1ED]' : 'text-[#292824]'}`}>채널·랭킹</button>
                      <button onClick={() => setActiveTab('voc')} className={`py-2 rounded-lg text-xs font-bold ${activeTab === 'voc' ? 'bg-[#866143] text-[#F2F1ED]' : 'text-[#292824]'}`}>리뷰 빈틈</button>
                      <button onClick={() => setActiveTab('ai')} className={`py-2 rounded-lg text-xs font-bold ${activeTab === 'ai' ? 'bg-[#866143] text-[#F2F1ED]' : 'text-[#292824]'}`}>AI 대본</button>
                      <button onClick={() => setActiveTab('funnel')} className={`py-2 rounded-lg text-xs font-bold ${activeTab === 'funnel' ? 'bg-[#866143] text-[#F2F1ED]' : 'text-[#292824]'}`}>퍼널 전략</button>
                    </div>

                    {activeTab === 'market' && (
                      <div className="space-y-4">
                        <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
                          {Object.entries(activeRef.channelMetrics).map(([k, v]) => (
                            <div key={k} className="p-3 rounded-xl bg-[#EAE6DF] border border-[#D5CFC4] flex justify-between items-center">
                              <div>
                                <div className="text-xs font-bold text-[#292824]">{v.label}</div>
                                <div className="text-[11px] text-[#292824]/70">{v.sub}</div>
                              </div>
                              <div className="text-right">
                                <div className="text-sm font-extrabold text-[#292824]">{v.value}</div>
                                <div className="text-[11px] font-bold text-[#866143]">{v.growth} ▲</div>
                              </div>
                            </div>
                          ))}
                        </div>
                        <div className="p-4 rounded-2xl bg-[#EAE6DF] border border-[#D5CFC4] space-y-2">
                          <div className="text-xs font-extrabold text-[#292824] mb-2">{activeRef.platformRankingTitle}</div>
                          {activeRef.rankings.map((r) => (
                            <div key={r.rank} className="p-2.5 rounded-xl bg-[#F2F1ED] border border-[#D5CFC4] flex justify-between items-center text-xs">
                              <div>
                                <span className="font-extrabold text-[#866143] mr-2">{r.rank}위</span>
                                <span className="font-bold text-[#292824]">{r.name}</span>
                                <span className="ml-2 text-[11px] text-[#866143]">★ {r.rating} ({r.reviews})</span>
                              </div>
                              <span className="font-extrabold shrink-0">{r.price}</span>
                            </div>
                          ))}
                        </div>
                      </div>
                    )}

                    {activeTab === 'voc' && (
                      <div className="space-y-3">
                        {activeRef.vocPainPoints.map((v, i) => (
                          <div key={i} className="p-4 rounded-2xl bg-[#EAE6DF] border border-[#D5CFC4]">
                            <div className="flex justify-between text-xs font-bold mb-1">
                              <span>소비자 불만 #{i + 1}</span>
                              <span className="text-[#866143]">언급 비중 {v.ratio}%</span>
                            </div>
                            <p className="text-sm font-extrabold text-[#292824] mb-2">“{v.issue}”</p>
                            <div className="w-full h-2 rounded-full bg-[#F2F1ED] overflow-hidden">
                              <div style={{ width: `${v.ratio}%` }} className="h-full bg-[#866143]" />
                            </div>
                          </div>
                        ))}
                      </div>
                    )}

                    {activeTab === 'ai' && (
                      <div className="space-y-3">
                        <div className="p-4 rounded-2xl bg-[#EAE6DF] border border-[#D5CFC4]">
                          <div className="grid grid-cols-4 gap-1 mb-2">
                            {['naver', 'reels', 'youtube', 'tiktok'].map((ch) => (
                              <button key={ch} onClick={() => setScriptChannelTab(ch)} className={`py-1 rounded text-[11px] font-bold uppercase ${scriptChannelTab === ch ? 'bg-[#866143] text-[#F2F1ED]' : 'bg-[#F2F1ED] text-[#292824]'}`}>{ch}</button>
                            ))}
                          </div>
                          <div className="p-3 rounded-xl bg-[#F2F1ED] text-xs whitespace-pre-line leading-relaxed">{activeRef.channelScripts[scriptChannelTab]}</div>
                        </div>
                        {activeRef.aiStrategies.map((st, i) => (
                          <div key={i} className="p-3.5 rounded-2xl bg-[#EAE6DF] border border-[#D5CFC4] flex justify-between items-center gap-3">
                            <p className="text-xs font-bold text-[#292824]">{st}</p>
                            <button onClick={() => handleCopyStrategy(st, i)} className="px-2.5 py-1.5 rounded-lg bg-[#F2F1ED] border border-[#D5CFC4] text-xs font-bold shrink-0">
                              {copiedIndex === i ? '복사됨!' : '복사'}
                            </button>
                          </div>
                        ))}
                      </div>
                    )}

                    {activeTab === 'funnel' && (
                      <div className="p-4 rounded-2xl bg-[#EAE6DF] border border-[#866143] text-xs leading-relaxed text-[#292824] font-medium">
                        <div className="font-extrabold text-[#866143] mb-2">크로스 채널 마케팅 퍼널 설계</div>
                        {activeRef.crossChannelFunnel}
                      </div>
                    )}
                  </div>
                </div>
              </div>
            </div>
          )}
        </div>
      );
    }

    const root = ReactDOM.createRoot(document.getElementById('root'));
    root.render(<App />);
